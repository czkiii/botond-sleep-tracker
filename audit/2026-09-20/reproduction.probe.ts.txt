const databases: DatabaseSync[] = []

function fixture(provider = 'APPLE') {
  const db = new DatabaseSync(':memory:'); databases.push(db)
  for (const path of ['schema.sql', 'migrations/003_accounts_and_sessions.sql', 'migrations/006_subscriptions_and_entitlements.sql', 'migrations/007_store_billing_state.sql']) db.exec(readFileSync(`worker/${path}`, 'utf8'))
  db.exec(`INSERT INTO accounts VALUES ('acc_a', NULL, NULL, NULL, 'hu', 'ACTIVE', 1, 1, 1, NULL)`)
  const binding = sqliteBinding(db)
  const service = new StoreBillingService(binding, { appleAccountToken: () => appleToken, googleAccountToken: () => googleToken, id: prefix => `${prefix}_${crypto.randomUUID()}` })
  const snapshot: any = { provider, environment: 'SANDBOX', providerSubscriptionId: 'sub-original', providerTransactionId: 'transaction-1', externalAccountToken: provider === 'APPLE' ? appleToken : googleToken, product: 'FAMILY_PLUS', status: 'ACTIVE', autoRenews: true, startedAt: now - 1000, trialEndsAt: null, currentPeriodEndsAt: now + 100_000, accessUntil: now + 100_000, verifiedAt: now, replacementProviderSubscriptionId: null }
  const event = (s: any, patch = {}) => ({ provider, providerEventId: crypto.randomUUID(), providerSubscriptionId: s.providerSubscriptionId, eventType: 'UPDATED', source: 'SERVER_NOTIFICATION', occurredAt: s.verifiedAt, receivedAt: s.verifiedAt, payloadHash: crypto.randomUUID(), ...patch })
  const apply = (s: any, e = event(s)) => service.applyVerifiedSubscription({ accountId: 'acc_a', subscription: s, event: e })
  const features = () => db.prepare('SELECT feature_key FROM account_entitlements WHERE revoked_at IS NULL').all().map(r => r.feature_key)
  return { db, binding, service, snapshot, event, apply, features }
}
it('AUDIT: equal-time same-transaction ACTIVE resurrects a revoked subscription', async () => {
  const f = fixture(); await f.service.getOrCreateAccountLink('acc_a', 'APPLE', now)
  await f.apply({ ...f.snapshot, status: 'REVOKED' })
  expect(f.features()).toEqual([])
  await f.apply(f.snapshot)
  expect(f.features()).toContain('FAMILY_PLUS_INSIGHTS')
})
it('AUDIT: Google linked replacement leaves old Plus grant after Family replacement', async () => {
  const f = fixture('GOOGLE_PLAY'); await f.service.getOrCreateAccountLink('acc_a', 'GOOGLE_PLAY', now)
  await f.apply(f.snapshot)
  await f.apply({ ...f.snapshot, providerSubscriptionId: 'sub-new', providerTransactionId: 'transaction-2', product: 'FAMILY', replacementProviderSubscriptionId: 'sub-original', verifiedAt: now + 1 })
  expect(f.features()).toContain('FAMILY_PLUS_INSIGHTS')
  expect(f.db.prepare('SELECT COUNT(*) AS n FROM subscriptions WHERE status = \'ACTIVE\'').get()?.n).toBe(2)
})
it('AUDIT: same provider identity accepts a switch between production and sandbox', async () => {

  await f.apply({ ...f.snapshot, environment: 'PRODUCTION' })
  await f.apply({ ...f.snapshot, verifiedAt: now + 1 })

it('AUDIT: conflicting event arriving between read and batch changes rights but cannot update event', async () => {
  const f = fixture(); await f.service.getOrCreateAccountLink('acc_a', 'APPLE', now)
  await f.apply(f.snapshot)
  const incoming = { ...f.snapshot, status: 'REVOKED', verifiedAt: now + 1 }
  const e = f.event(incoming, { providerEventId: 'racing-event', payloadHash: 'incoming-hash' })
  const originalBatch = f.binding.batch.bind(f.binding)
  f.binding.batch = async (statements: any) => {
    const subId = f.db.prepare('SELECT id FROM subscriptions').get()?.id
    f.db.prepare(`INSERT INTO subscription_events VALUES (?, 'APPLE', 'racing-event', ?, 'OTHER', ?, ?, 'other-hash', ?, NULL)`).run('evt_raced', subId as string, now, now, now)
    return originalBatch(statements)
  }
  expect((await f.apply(incoming, e)).outcome).toBe('APPLIED')
  expect(f.features()).toEqual([])
  expect(f.db.prepare("SELECT payload_hash FROM subscription_events WHERE provider_event_id = 'racing-event'").get()?.payload_hash).toBe('other-hash')
})

async function familyFixture() {
  const f = fixture()
  f.db.exec(readFileSync('worker/migrations/005_family_memberships.sql', 'utf8'))
  const env = { DB: f.binding, TOKEN_PEPPER: 'audit-only', ALLOWED_ORIGINS: '', ENTITLEMENT_ENFORCEMENT: 'true' }
  const create = await worker.fetch(new Request('https://local.test/v1/families', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ familyName: 'Audit', childId: 'child-a' }) }), env)
  expect(create.status).toBe(201)
  const family: any = (await create.json()).data
  const req = (path: string, method = 'GET', body?: unknown) => worker.fetch(new Request(`https://local.test${path}`, { method, headers: { Authorization: `Bearer ${family.deviceToken}`, 'Content-Type': 'application/json' }, ...(body ? { body: JSON.stringify(body) } : {}) }), env)
  return { ...f, env, family, req }
}

it('AUDIT: concurrent child deletes can both pass the last-child precheck', async () => {
  const f = await familyFixture()
  expect((await f.req('/v1/children', 'POST', { operationId: 'create-b', child: { id: 'child-b', name: 'B' } })).status).toBe(201)
  const originalBatch = f.binding.batch.bind(f.binding)
  let injected = false
  f.binding.batch = async (statements: any) => {
    if (!injected) { injected = true; expect((await f.req('/v1/children/child-b', 'DELETE', { operationId: 'delete-b' })).status).toBe(200) }
    return originalBatch(statements)
  }
  expect((await f.req('/v1/children/child-a', 'DELETE', { operationId: 'delete-a' })).status).toBe(200)
  expect(f.db.prepare('SELECT COUNT(*) AS n FROM children WHERE deleted_at IS NULL').get()?.n).toBe(0)
})

it('AUDIT: overlapping records yield different Today and statistics totals', () => {
  const a = createSession('a', '2026-09-19T12:00:00Z', '2026-09-19T13:00:00Z')
  const b = { ...a, id: 'b', startTime: '2026-09-19T12:30:00Z', endTime: '2026-09-19T13:30:00Z' }
  const clock = Date.parse('2026-09-19T15:00:00Z')
  expect(totalToday([a, b], new Date(clock))).toBe(120 * 60_000)
  expect(buildSleepDaySummaries([a, b], clock).reduce((sum, d) => sum + d.totalMs, 0)).toBe(90 * 60_000)
})
