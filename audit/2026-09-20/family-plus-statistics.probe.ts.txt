// Audit only: defect probes assert current behavior; controls use independent expected values.
// Copy to repository root as family-stats-audit.test.ts before running. See README.md.
import { beforeEach, afterEach, describe, it, expect } from 'vitest'
import { buildInsightsFoundation } from './src/insights'
import { buildPredictionLite } from './src/prediction'
import { buildSimilarDaysInsight } from './src/similarDays'
import { buildSleepDaySummaries, buildSleepDevelopment } from './src/sleepDevelopment'
import { buildSleepChangeInsight } from './src/sleepChange'
import { buildMonthlyFamilyReport } from './src/monthlyReport'
import { splitDayNight, getDataQualityReport, formatDuration } from './src/utils'
import type { SleepSession } from './src/types'

const H = 3600_000, M = 60_000
const originalTZ = process.env.TZ
beforeEach(() => { process.env.TZ = 'UTC' })
afterEach(() => { if (originalTZ === undefined) delete process.env.TZ; else process.env.TZ = originalTZ })
function at(month: number, day: number, hour = 0, minute = 0) { return Date.UTC(2026, month - 1, day, hour, minute) }
function s(id: string, start: number, end: number | null, kind: SleepSession['dayNightOverride'] = null): SleepSession {
  const startTime = new Date(start).toISOString(), endTime = end === null ? null : new Date(end).toISOString()
  return { id, childId: 'audit-child', startTime, endTime, note: '', dayNightOverride: kind, createdAt: startTime, updatedAt: endTime ?? startTime }
}
function daily(month: number, days: number[], startHour: number, hours: number, kind: SleepSession['dayNightOverride'] = null) {
  return days.map(d => s(`${month}-${d}-${startHour}`, at(month, d, startHour), at(month, d, startHour) + hours * H, kind))
}
const seq = (n: number, start = 1) => Array.from({ length: n }, (_, i) => i + start)

describe('independent controls', () => {
  it('median and interpolated quartiles of gaps [1,2,3,4] hours', () => {
    let cursor = at(1, 20, 1)
    const data = [s('0', cursor, cursor + H)]
    for (const gap of [1, 2, 3, 4]) { cursor += H + gap * H; data.push(s(String(gap), cursor, cursor + H)) }
    const result = buildInsightsFoundation(data, at(1, 21, 12)).wakeWindow
    expect(result.sampleCount).toBe(4)
    expect(result.typicalMs).toBe(2.5 * H)
    expect(result.typicalRange).toEqual({ lowMs: 1.75 * H, highMs: 3.25 * H })
    expect(buildInsightsFoundation([...data].reverse(), at(1, 21, 12))).toEqual(buildInsightsFoundation(data, at(1, 21, 12)))
  })
  it('interval union agrees with independent occupied-minute oracle across 50 seeded datasets', () => {
    let seed = 923; const random = () => { seed = (seed * 1664525 + 1013904223) >>> 0; return seed }
    for (let run = 0; run < 50; run++) {
      const occupied = new Set<number>(), data: SleepSession[] = []
      for (let i = 0; i < 15; i++) {
        const start = random() % 500, length = 2 + random() % 100
        for (let j = start; j < start + length; j++) occupied.add(j)
        data.push(s(`r${run}-${i}`, at(1, 1, 7) + start * M, at(1, 1, 7) + (start + length) * M, 'day'))
      }
      const result = buildSleepDaySummaries(data, at(1, 2))
      expect(result.reduce((sum, day) => sum + day.totalMs, 0)).toBe(occupied.size * M)
      expect(buildSleepDaySummaries([...data].reverse(), at(1, 2))).toEqual(result)
      expect(buildSleepDaySummaries([...data, ...data], at(1, 2))).toEqual(result)
    }
  })
  it.each([
    ['2026-03-28T23:00:00Z', '2026-03-29T06:00:00Z', 7],
    ['2026-10-24T22:00:00Z', '2026-10-25T07:00:00Z', 9]
  ])('Budapest DST conserves elapsed duration: %s', (start, end, hours) => {
    process.env.TZ = 'Europe/Budapest'
    const item = s('dst', Date.parse(String(start)), Date.parse(String(end)))
    const days = buildSleepDaySummaries([item], Date.parse(String(end)) + H)
    expect(days).toHaveLength(1)
    expect(days[0].totalMs).toBe(Number(hours) * H)
    expect(days[0].dayMs + days[0].nightMs).toBe(days[0].totalMs)
  })
  it('change detection is neutral for identical sufficiently recorded periods', () => {
    const data = [...daily(1, seq(14, 10), 0, 10), ...daily(2, seq(5, 5), 0, 10)]
    expect(buildSleepChangeInsight(data, at(2, 10))).toMatchObject({ status: 'stable', recentSampleCount: 5, baselineSampleCount: 14, signals: [] })
  })
  it('monthly baseline is the median of monthly means; +60min night is exact', () => {
    const data = [...daily(4, seq(14), 0, 8, 'night'), ...daily(5, seq(14), 0, 8, 'night'), ...daily(6, seq(14), 0, 8, 'night'), ...daily(7, seq(14), 0, 9, 'night')]
    const result = buildMonthlyFamilyReport(data, at(8, 20))
    expect(result.month?.key).toBe('2026-07')
    expect(result.baselineMonthCount).toBe(3)
    expect(result.trends).toContainEqual({ metric: 'night', direction: 'higher', baselineValue: 8 * H, currentValue: 9 * H, delta: H })
  })
  it('5 minute record is retained; weekly division and flooring explain display of 0 minutes', () => {
    const data = [s('five', at(1, 1, 7), at(1, 1, 7, 5))]
    expect(buildSleepDaySummaries(data, at(1, 2))[0].totalMs).toBe(5 * M)
    expect(formatDuration(5 * M / 7, 'hu')).toBe('0 p')
  })
})

describe('reproduced discrepancies and evidence limits', () => {
  it('S01: minute stepping misallocates a seconds-offset day/night boundary', () => {
    const item = s('edge', at(1, 1, 18, 59) + 30_000, at(1, 1, 19, 2) + 30_000)
    expect(splitDayNight(item)).toEqual({ day: M, night: 2 * M })
    const exact = buildSleepDaySummaries([item], at(1, 2))[0]
    expect(exact.dayMs).toBe(30_000); expect(exact.nightMs).toBe(150_000)
  })
  it('S02: fragmented nights select a 01:00 wake instead of final 06:00', () => {
    const data = seq(3, 9).flatMap(d => [s(`early${d}`, at(1, d, 20), at(1, d + 1, 1)), s(`late${d}`, at(1, d + 1, 1, 30), at(1, d + 1, 6))])
    const result = buildInsightsFoundation(data, at(1, 12, 12)).routine
    expect(result.wakeTime?.typicalMinutes).toBe(60)
    expect(result.wakeTime?.sampleCount).toBe(4)
    expect(result.observedDayCount).toBe(3)
  })
  it('S03: wake times around midnight yield a noon linear median', () => {
    const data = seq(4).map((_, i) => {
      const d = 1 + i * 2
      return s(`midnight${i}`, at(1, d, 20), i % 2 ? at(1, d + 1, 0, 5) : at(1, d, 23, 55), 'night')
    })
    expect(buildInsightsFoundation(data, at(1, 10, 12)).routine.wakeTime?.typicalMinutes).toBe(720)
  })
  it('S04: one wake sample still has a displayable typical value', () => {
    const result = buildInsightsFoundation([s('a', at(1, 1, 8), at(1, 1, 9)), s('b', at(1, 1, 11), at(1, 1, 12))], at(1, 1, 13)).wakeWindow
    expect(result).toMatchObject({ status: 'collecting', sampleCount: 1, typicalMs: 2 * H })
    expect(result.typicalRange).toEqual({ lowMs: 2 * H, highMs: 2 * H })
  })
  it('S05: sample-count confidence can reach medium using just one day', () => {
    const data = seq(8, 0).map(i => s(String(i), at(1, 1, 6) + i * 90 * M, at(1, 1, 6) + (i * 90 + 30) * M))
    expect(buildInsightsFoundation(data, at(1, 1, 19)).wakeWindow).toMatchObject({ sampleCount: 7, confidence: 'medium' })
  })
  it('S06: old history outside selected 7 days changes prediction bucket', () => {
    const recent = seq(3, 26).flatMap(d => [s(`r${d}a`, at(1, d, 8), at(1, d, 9)), s(`r${d}b`, at(1, d, 13), at(1, d, 14))])
    recent.push(s('today', at(1, 30, 8), at(1, 30, 9)))
    expect(buildPredictionLite(recent, at(1, 30, 12), 7).bucket).toBe('day-2')
    const old = daily(1, seq(20), 8, 1)
    expect(buildPredictionLite([...old, ...recent], at(1, 30, 12), 7).bucket).toBe('night')
  })
  it('S07: prediction remains ready using a last wake more than two days old', () => {
    const data = seq(3, 20).flatMap(d => [s(`n${d}`, at(1, d - 1, 20), at(1, d, 6)), s(`d${d}`, at(1, d, 8), at(1, d, 9))])
    data.push(s('stale-wake', at(1, 27, 20), at(1, 28, 6)))
    const result = buildPredictionLite(data, at(1, 30, 9), 14)
    expect(result.status).toBe('ready'); expect(result.windowEnd).toBe(at(1, 28, 8))
    expect(result.currentWakeMs).toBe(51 * H)
  })
  it('S08: similar days can report ready during an active sleep', () => {
    const data = daily(1, seq(4, 17), 8, 1)
    data.push(s('active', at(1, 20, 11), null))
    const result = buildSimilarDaysInsight(data, at(1, 20, 12))
    expect(result.status).toBe('ready')
    expect(result.current?.awakeMs).toBe(3 * H)
  })
  it('S09: every valid historical day is a candidate even when very dissimilar', () => {
    const data = daily(1, seq(3, 17), 0, 11)
    data.push(s('today', at(1, 20, 8), at(1, 20, 9)))
    const result = buildSimilarDaysInsight(data, at(1, 20, 12))
    expect(result.status).toBe('ready')
    expect(result.matches).toHaveLength(3)
    expect(result.matches.every(m => m.differences.totalSleepMs === 10 * H)).toBe(true)
  })
  it('S10: sparse overnight records halve monthly longest versus development', () => {
    const data = [5, 6].flatMap(month => seq(7).map((_, i) => s(`${month}-${i}`, at(month, 1 + i * 2, 20), at(month, 2 + i * 2, 8), 'night')))
    const development = buildSleepDevelopment(data, at(7, 10)).latest!
    const report = buildMonthlyFamilyReport(data, at(7, 10)).month!
    expect(development.recordedDays).toBe(14); expect(report.recordedDays).toBe(14)
    expect(development.averageLongestBlockMs).toBe(12 * H)
    expect(report.averageLongestBlockMs).toBe(6 * H)
  })
  it('S11: four recent partially logged days are enough for a strong change claim', () => {
    const data = [...daily(1, seq(14, 10), 0, 10), ...daily(2, seq(4, 5), 12, 1)]
    const result = buildSleepChangeInsight(data, at(2, 10))
    expect(result).toMatchObject({ status: 'changed', recentSampleCount: 4, baselineSampleCount: 14 })
    expect(result.signals.some(x => x.severity === 'strong')).toBe(true)
  })
  it('S12: latest usable monthly report can be months older than latest closed month', () => {
    const result = buildMonthlyFamilyReport([...daily(5, seq(14), 0, 8), ...daily(6, seq(14), 0, 8)], at(9, 20))
    expect(result.status).toBe('ready'); expect(result.month?.key).toBe('2026-06')
  })
  it('S13: overlapping opposite overrides are excluded in wake analytics but counted as night in development', () => {
    const a = s('a', at(1, 1, 12), at(1, 1, 14), 'day'), b = { ...a, id: 'b', dayNightOverride: 'night' as const }
    expect(getDataQualityReport([a, b], at(1, 2)).excludedSessionIds).toHaveLength(2)
    expect(buildInsightsFoundation([a, b], at(1, 2)).quality.usableSessionCount).toBe(0)
    expect(buildSleepDaySummaries([a, b], at(1, 2))[0].nightMs).toBe(2 * H)
  })
  it('S14: device timezone changes number of recorded dates for same absolute sleep', () => {
    const data = [s('tz', at(1, 1, 21, 30), at(1, 1, 23, 30))]
    expect(buildSleepDaySummaries(data, at(1, 3))).toHaveLength(1)
    process.env.TZ = 'Europe/Budapest'
    expect(buildSleepDaySummaries(data, at(1, 3))).toHaveLength(2)
    process.env.TZ = 'Asia/Tokyo'
    expect(buildSleepDaySummaries(data, at(1, 3))[0].key).toBe('2026-01-02')
  })
  it('S15: before-night sample bucket includes brief awakenings within the night', () => {
    const data = seq(3, 10).flatMap(d => [
      s(`nap${d}`, at(1, d, 12), at(1, d, 13)),
      s(`n1${d}`, at(1, d, 20), at(1, d + 1, 0)),
      s(`n2${d}`, at(1, d + 1, 0, 15), at(1, d + 1, 3)),
      s(`n3${d}`, at(1, d + 1, 3, 15), at(1, d + 1, 6))
    ])
    const night = buildInsightsFoundation(data, at(1, 13, 12)).wakeWindow.breakdown.find(b => b.key === 'night')!
    expect(night.sampleCount).toBe(9)
    expect(night.typicalMs).toBe(15 * M)
  })
})
